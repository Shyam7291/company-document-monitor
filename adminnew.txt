        <section className="sr-section"><div className="sr-section-head"><h2>Manage Operations</h2><span>Quick access</span></div><div className="sr-operations">{operations.map(o => <button key={o.title} className={`sr-operation ${o.color}`} onClick={() => showToast(`Opening ${o.title}`)}><span className="sr-op-icon"><Icon name={o.icon}/></span><span><h3>{o.title}</h3><p>{o.text}</p></span><span className="sr-op-arrow"><Icon name="chevron" size={17}/></span></button>)}</div></section>
      </div>
    </main>

    <nav className="sr-bottom" aria-label="Primary navigation">{[["Home","home"],["Orders","order"],["Samples","sample"]].map(([label,icon]) => <button key={label} className={`sr-nav-item ${activeNav===label?"active":""}`} onClick={() => navClick(label)}><Icon name={icon} size={18}/><span>{label}</span></button>)}</nav>

    {drawer && <><div className="sr-backdrop" onClick={() => setDrawer(false)}/><aside className="sr-drawer" role="dialog" aria-modal="true" aria-label="Admin menu"><div className="sr-drawer-head"><span className="sr-avatar">AK</span><div><h2>Alok</h2><p>StoneRate Administrator</p></div><ButtonIcon label="Close menu" icon="close" onClick={() => setDrawer(false)}/></div><div className="sr-menu">{menu.map(([label,desc,icon]) => <button key={label} className={`sr-menu-item ${activeMenu===label?"active":""} ${label==="Logout"?"logout":""}`} onClick={() => menuClick(label)}><span className="sr-menu-icon"><Icon name={icon} size={17}/></span><span className="sr-menu-copy"><strong>{label}</strong><span>{desc}</span></span></button>)}</div></aside></>}

    {notifications && <><div className="sr-backdrop" onClick={() => setNotifications(false)}/><section className="sr-sheet" role="dialog" aria-modal="true" aria-label="Notifications"><div className="sr-sheet-head"><div><h2>Notifications</h2><p>Recent StoneRate operational alerts</p></div><ButtonIcon label="Close notifications" icon="close" onClick={() => setNotifications(false)}/></div><div className="sr-notice-list">{notices.map(([title,text,isUnread,icon]) => <button className={`sr-notice ${isUnread?"unread":""}`} key={title} style={{width:"100%",textAlign:"left",cursor:"pointer"}} onClick={() => showToast(`Opening ${title}`)}><span className="sr-notice-icon"><Icon name={icon} size={17}/></span><span><h3>{title}</h3><p>{text}</p></span>{isUnread && <i className="sr-unread-dot"/>}</button>)}</div></section></>}

    {logout && <><div className="sr-backdrop" onClick={() => setLogout(false)}/><div className="sr-modal-wrap"><div className="sr-modal" role="alertdialog" aria-modal="true" aria-labelledby="logout-title"><span className="sr-modal-icon"><Icon name="logout"/></span><h2 id="logout-title">Sign out of StoneRate?</h2><p>You will need to sign in again to access the operations dashboard.</p><div className="sr-modal-actions"><button className="sr-btn" onClick={() => setLogout(false)}>Cancel</button><button className="sr-btn danger" onClick={() => { setLogout(false); setDrawer(false); showToast("Logout confirmed (demo)"); }}>Sign out</button></div></div></div></>}
    {toast && <div className="sr-toast" role="status">{toast}</div>}
  </div>;
}
