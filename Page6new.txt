function useViewport() {
  const [width, setWidth] = useState(() =>
    typeof window === "undefined" ? 1280 : window.innerWidth
  );
  const [height, setHeight] = useState(() =>
    typeof window === "undefined" ? 900 : window.innerHeight
  );

  useEffect(() => {
    if (typeof window === "undefined") return undefined;
    const handleResize = () => {
      setWidth(window.innerWidth);
      setHeight(window.innerHeight);
    };
    window.addEventListener("resize", handleResize);
    return () => window.removeEventListener("resize", handleResize);
  }, []);

  const desktop = width >= 700;
  return { width, height, desktop, mobile: !desktop };
}

function getIndiaTodayParts() {
  const now = new Date();
  const ist = new Date(now.getTime() + now.getTimezoneOffset() * 60000 + 330 * 60000);
  return { year: ist.getUTCFullYear(), month: ist.getUTCMonth() + 1, day: ist.getUTCDate() };
}

function pad2(n) {
  return String(n).padStart(2, "0");
}

function parseDateKey(value) {
  if (typeof value !== "string" || !value) return null;
  const parts = value.split("-");
  if (parts.length !== 3) return null;
  const [y, m, d] = parts.map((p) => Number(p));
  if (!Number.isInteger(y) || !Number.isInteger(m) || !Number.isInteger(d)) return null;
  if (m < 1 || m > 12 || d < 1 || d > 31) return null;
  return { year: y, month: m, day: d };
}

function isNotPastDate(value) {
  const parts = parseDateKey(value);
  if (!parts) return false;
  const t = getIndiaTodayParts();
  const startOfToday = new Date(t.year, t.month - 1, t.day).getTime();
  const selected = new Date(parts.year, parts.month - 1, parts.day).getTime();
  return selected >= startOfToday;
}

function isMaterialExpired(material) {
  const parts = parseDateKey(material && material.sampleExpiresAt);
  if (!parts) return false;
  const t = getIndiaTodayParts();
  const startOfToday = new Date(t.year, t.month - 1, t.day).getTime();
  const expiry = new Date(parts.year, parts.month - 1, parts.day).getTime();
  return expiry < startOfToday;
}

function formatDateDisplay(value) {
  const parts = parseDateKey(value);
  if (!parts) return value || "";
  const d = new Date(parts.year, parts.month - 1, parts.day);
  if (Number.isNaN(d.getTime())) return value;
  return d.toLocaleDateString("en-IN", {
    weekday: "short",
    day: "numeric",
    month: "short",
    year: "numeric",
  });
}

function formatDateTime(value) {
  if (!value) return "Not provided";
  const d = new Date(value);
  if (Number.isNaN(d.getTime())) return String(value);
  return d.toLocaleString("en-IN", {
    day: "numeric",
    month: "short",
    year: "numeric",
    hour: "numeric",
    minute: "2-digit",
  });
}

function normalizeStoredContact(value) {
  const digits = String(value || "").replace(/\D/g, "");
  if (digits.length > 10 && digits.slice(0, 2) === "91") {
    return digits.slice(2).slice(0, 10);
  }
  return digits.slice(0, 10);
}

function normalizeMaterials(items) {
  const result = [];
  items.forEach((raw) => {
    if (!raw || typeof raw !== "object") return;
    const rawVehicles = Array.isArray(raw.vehicles) ? raw.vehicles : [];
    const vehicles = rawVehicles
      .map((v) => {
        const quantity = Number(v && v.quantity) || 0;
        const capacityTons = Number(v && v.capacityTons) || 0;
        const totalTons = quantity > 0 ? capacityTons * quantity : 0;
        return {
          vehicleId: v && v.vehicleId !== undefined && v.vehicleId !== null ? v.vehicleId : "",
          vehicleName: v && v.vehicleName ? v.vehicleName : "Vehicle",
          capacityTons,
          quantity,
          totalTons: Math.round(totalTons * 100) / 100,
        };
      })
      .filter((v) => v.quantity > 0);
    if (vehicles.length === 0) return;
    const totalVehicles = vehicles.reduce((sum, v) => sum + v.quantity, 0);
    const totalTons =
      Math.round(vehicles.reduce((sum, v) => sum + v.totalTons, 0) * 100) / 100;
    result.push({
      sampleId: raw.sampleId !== undefined && raw.sampleId !== null ? raw.sampleId : "",
      materialName: raw.materialName || "Material",
      sampleCode: raw.sampleCode !== undefined && raw.sampleCode !== null ? raw.sampleCode : "",
      sampleImageUrl: raw.sampleImageUrl || "",
      sampleSourceArea: raw.sampleSourceArea || "",
      sampleUploadedAt: raw.sampleUploadedAt || "",
      sampleExpiresAt: raw.sampleExpiresAt || "",
      sampleAdminNote: raw.sampleAdminNote || "",
      vehicles,
      totalVehicles,
      totalTons,
      expired: isMaterialExpired(raw),
    });
  });
  return result;
}

function computeTotals(materials) {
  return {
    totalMaterials: materials.length,
    totalVehicles: materials.reduce((sum, m) => sum + (m.totalVehicles || 0), 0),
    totalTons:
      Math.round(materials.reduce((sum, m) => sum + (m.totalTons || 0), 0) * 100) / 100,
  };
}

function buildValidation(date, contact, address) {
  const errors = {};
  if (!date) {
    errors.requestedArrivalDate = "Choose a requested arrival date.";
  } else if (!isNotPastDate(date)) {
    errors.requestedArrivalDate = "Delivery date cannot be in the past.";
  }
  if (!contact || contact.length !== 10) {
    errors.contactNumber = "Enter the complete 10-digit mobile number.";
  }
  const trimmedAddress = String(address || "").trim();
  if (!trimmedAddress) {
    errors.deliveryAddress = "Enter the delivery address.";
  } else if (trimmedAddress.length < 10) {
    errors.deliveryAddress = "Delivery address must be at least 10 characters.";
  } else if (trimmedAddress.length > 1000) {
    errors.deliveryAddress = "Delivery address must be under 1,000 characters.";
  }
  return errors;
}

        <Icon name="arrowRight" size={15} color="#171006" strokeWidth={2.4} />
        <span>Return to Today's Materials</span>
      </button>
    </div>
  );
}

  const doneRef = useRef(null);

  useEffect(() => {
    if (doneRef.current) doneRef.current.focus();
  }, []);

  return (
    <div style={styles.popupOverlay} className="p6-fade">
      <div
        role="dialog"
        aria-modal="true"
        aria-labelledby="p6-success-title"
        aria-describedby="p6-success-text"
        style={styles.popupCard}
        className="p6-pop"
      >
        <div style={styles.popupGlow} aria-hidden="true" />
        <div style={styles.popupSymbolWrap} aria-hidden="true">
          <span className="p6-ring-pulse" />
          <span className="p6-ring-pulse" style={{ animationDelay: "0.6s" }} />
          <div style={styles.popupSymbol}>
            <svg
              width="42"
              height="42"
              viewBox="0 0 24 24"
              fill="none"
              stroke="#04210f"
              strokeWidth="3"
              strokeLinecap="round"
              strokeLinejoin="round"
              focusable="false"
            >
              <path className="p6-draw" d="M20 6L9 17l-5-5" />

          </div>
        </div>
        <h2 id="p6-success-title" style={styles.popupTitle}>
          Rate Request Submitted
        </h2>
        {request && request.publicRequestId ? (
          <div style={styles.popupId}>Request ID: {request.publicRequestId}</div>
        ) : null}
        <div style={styles.popupStatus}>
          <span style={styles.popupPulseDot} aria-hidden="true" />
          <span>Checking sellers</span>
        </div>
        <p id="p6-success-text" style={styles.popupText}>
          StoneRate has started checking material-wise rates and transport availability.
        </p>
        <button

          style={styles.popupDone}
          className="p6-focusable p6-cta p6-green"
        >
          Done
        </button>
      </div>
    </div>
  );
}

export default function OrderConfirmationPage({ orderDraft, goToPage5, goToPage7 }) {
  const viewport = useViewport();
  const styles = useMemo(() => createStyles(viewport), [viewport.width, viewport.height]);

  const orderItems = Array.isArray(orderDraft?.materials)
    ? orderDraft.materials
    : [];

  const normalizedMaterials = useMemo(() => normalizeMaterials(orderItems), [orderItems]);
  const totals = useMemo(() => computeTotals(normalizedMaterials), [normalizedMaterials]);
  const { totalMaterials, totalVehicles, totalTons } = totals;

  const todayKey = useMemo(() => {
    const t = getIndiaTodayParts();
    return `${t.year}-${pad2(t.month)}-${pad2(t.day)}`;
  }, []);

  const [requestedArrivalDate, setRequestedArrivalDate] = useState(
    () => (orderDraft && orderDraft.requestedArrivalDate) || ""
  );
  const [contactNumber, setContactNumber] = useState(() =>
    normalizeStoredContact(orderDraft && orderDraft.contactNumber)
  );
  const [deliveryAddress, setDeliveryAddress] = useState(
    () => (orderDraft && orderDraft.deliveryAddress) || ""
  );
  const [orderNotes, setOrderNotes] = useState(
    () => (orderDraft && orderDraft.notes) || ""
  );
  const [confirmed, setConfirmed] = useState(false);
  const [submitted, setSubmitted] = useState(false);
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [submitError, setSubmitError] = useState(null);
  const [createdRequest, setCreatedRequest] = useState(null);
  const [previewItem, setPreviewItem] = useState(null);
  const [validationErrors, setValidationErrors] = useState({});

  const liveErrors = useMemo(
    () => buildValidation(requestedArrivalDate, contactNumber, deliveryAddress),
    [requestedArrivalDate, contactNumber, deliveryAddress]
  );

  const hasExpiredSample = normalizedMaterials.some((m) => m.expired);
  const hasMaterials = normalizedMaterials.length > 0;

  const canSubmit =
    hasMaterials &&
    !hasExpiredSample &&
    !liveErrors.requestedArrivalDate &&
    !liveErrors.contactNumber &&
    !liveErrors.deliveryAddress &&
    confirmed &&
    !isSubmitting &&
    !submitted;

  const completeness = [
    hasMaterials && !hasExpiredSample,
    !!requestedArrivalDate && !liveErrors.requestedArrivalDate,
    contactNumber.length === 10,
    deliveryAddress.trim().length >= 10,
    confirmed,
  ].filter(Boolean).length;
  const progressPct = Math.round((completeness / 5) * 100);

  useEffect(() => {
    if (document.getElementById("page6-global-css")) return;
    const styleEl = document.createElement("style");
    styleEl.id = "page6-global-css";
    styleEl.textContent = globalCss;
    document.head.appendChild(styleEl);
  }, []);

  const handleDateChange = (e) => {
    setRequestedArrivalDate(e.target.value);
    setValidationErrors((prev) => ({ ...prev, requestedArrivalDate: undefined }));
  };
  const handleDateBlur = () => {
    setValidationErrors((prev) => ({
      ...prev,
      requestedArrivalDate: liveErrors.requestedArrivalDate,
    }));
  };
  const handleContactChange = (e) => {
    const digits = e.target.value.replace(/\D/g, "").slice(0, 10);
    setContactNumber(digits);
    setValidationErrors((prev) => ({ ...prev, contactNumber: undefined }));
  };
  const handleContactBlur = () => {
    setValidationErrors((prev) => ({
      ...prev,
      contactNumber: liveErrors.contactNumber,
    }));
  };
  const handleAddressChange = (e) => {
    setDeliveryAddress(e.target.value);
    setValidationErrors((prev) => ({ ...prev, deliveryAddress: undefined }));
  };
  const handleAddressBlur = () => {
    setValidationErrors((prev) => ({
      ...prev,
      deliveryAddress: liveErrors.deliveryAddress,
    }));
  };
  const handleNotesChange = (e) => {
    setOrderNotes(e.target.value.slice(0, 2000));
  };

  const handleSubmit = async (e) => {
    if (e && e.preventDefault) e.preventDefault();
    if (!canSubmit) return;
    setValidationErrors({});
    setIsSubmitting(true);
    setSubmitError(null);
    const completeOrderDraft = {
      ...orderDraft,
      materials: normalizedMaterials,
      totalMaterials,
      totalVehicles,
      totalTons,
      requestedArrivalDate,
      contactNumber,
      deliveryAddress,
      notes: orderNotes.trim(),
    };
    try {
      const result = await createRateRequest(completeOrderDraft);
      setCreatedRequest(result);
      setSubmitted(true);
    } catch (error) {
      setSubmitError(
        (error && error.message) ||
          "We could not submit your rate request. Check your connection and try again."
      );
    } finally {
      setIsSubmitting(false);
    }
  };

  const submitButtonStyle = mergeStyles(
    styles.submitButton,
    !canSubmit ? styles.submitButtonDisabled : null
  );

  return (
    <div style={styles.page}>
      <div style={styles.frame} role="region" aria-label="Rate request confirmation">
        <div style={styles.gridOverlay} aria-hidden="true" />
        <div style={styles.stars} className="p6-stars" aria-hidden="true" />
        <div style={styles.orbBlue} className="p6-aurora-a" aria-hidden="true" />
        <div style={styles.orbViolet} className="p6-aurora-b" aria-hidden="true" />
        <div style={styles.orbGold} className="p6-aurora-c" aria-hidden="true" />

        <header style={styles.header}>
          <div style={styles.topBar}>
            <button
              type="button"
              onClick={goToPage5}
              disabled={isSubmitting}
              style={mergeStyles(styles.iconBtn, isSubmitting ? styles.iconBtnDisabled : {})}
              className="p6-focusable p6-ghost"
              aria-label="Back to review"
            >
              <Icon name="chevronLeft" size={16} />
              <span>Back</span>
            </button>
            <button
              type="button"
              onClick={goToPage5}
              disabled={isSubmitting}
              style={mergeStyles(styles.iconBtn, isSubmitting ? styles.iconBtnDisabled : {})}
              className="p6-focusable p6-ghost"
            >
              <Icon name="edit" size={14} />
              Edit
            </button>
          </div>
          <div style={styles.badge}>
            <span style={styles.badgeDot} aria-hidden="true" />
            <span>Final Review</span>
          </div>
          <h1 style={styles.title}>Confirm Your Request</h1>
          <p style={styles.subtitle}>
            Review selected quality references and add delivery details.
          </p>
          <div style={styles.summaryGrid}>
            <SummaryMetric
              styles={styles}
              label="Materials"
              value={totalMaterials}
              icon={<Icon name="cube" size={19} color="#0b0f1c" strokeWidth={1.9} />}
              grad="linear-gradient(135deg, #5b8cff, #7aa0ff)"
            />
            <SummaryMetric
              styles={styles}
              label="Vehicles"
              value={totalVehicles}
              icon={<Icon name="truck" size={19} color="#0b0f1c" strokeWidth={1.9} />}
              grad="linear-gradient(135deg, #a78bfa, #8b5cf6)"
            />
            <SummaryMetric
              styles={styles}
              label="Tons"
              value={totalTons}
              unit="T"
              icon={<Icon name="activity" size={19} color="#171006" strokeWidth={2.2} />}
              grad="linear-gradient(135deg, #ffd479, #f5a623)"
            />
            <SummaryMetric
              styles={styles}
              label="Visual refs"
              value={totalMaterials}
              icon={<Icon name="image" size={18} color="#04211f" strokeWidth={1.9} />}
              grad="linear-gradient(135deg, #38e1d4, #22b8cf)"
            />
          </div>
        </header>

        <main style={styles.main} className="p6-scroll">
          {!hasMaterials ? (
            <EmptyDraftState styles={styles} onGoBack={goToPage5} />
          ) : (
            <>
              <div style={styles.sectionRow}>
                <span style={styles.sectionDot} aria-hidden="true" />
                <span style={styles.sectionLabel}>Selected materials</span>
                <span style={styles.countChip}>{normalizedMaterials.length}</span>
                <span style={styles.sectionLine} aria-hidden="true" />
              </div>
              {hasExpiredSample ? (
                <div role="alert" style={styles.expiredBanner}>
                  <Icon name="alert" size={16} color={C.red} />
                  <div style={styles.expiredBannerBody}>
                    <div style={styles.expiredBannerTitle}>Expired sample detected</div>
                    <div style={styles.expiredBannerText}>
                      This daily sample has expired. Return to the material gallery and select a
                      fresh reference.
                    </div>
                  </div>
                  <button
                    type="button"
                    onClick={goToPage5}
                    style={styles.expiredButton}
                    className="p6-focusable"
                  >
                    Return to gallery
                  </button>
                </div>
              ) : null}
              {normalizedMaterials.map((material, index) => (
                <div
                  key={material.sampleId || index}
                  className="p6-scale"
                  style={{ animationDelay: `${index * 80}ms` }}
                >
                  <SelectedMaterialCard
                    styles={styles}
                    material={material}
                    index={index}
                    busy={isSubmitting}
                    onViewImage={setPreviewItem}
                    onEdit={goToPage5}
                  />
                </div>
              ))}
            </>
          )}

          {hasMaterials ? (
            <>
              <form id="rate-request-form" noValidate onSubmit={handleSubmit}>
                <div style={styles.sectionRow}>
                  <span style={styles.sectionDot} aria-hidden="true" />
                  <span style={styles.sectionLabel}>Delivery details</span>
                  <span style={styles.sectionLine} aria-hidden="true" />
                </div>

                <DeliveryField
                  styles={styles}
                  label="Requested arrival date"
                  required
                  error={validationErrors.requestedArrivalDate}
                  icon={<Icon name="calendar" size={15} />}
                >
                  <input
                    type="date"
                    className="p6-input p6-grad-input p6-date"
                    style={styles.dateInput}
                    min={todayKey}
                    value={requestedArrivalDate}
                    onChange={handleDateChange}
                    onBlur={handleDateBlur}
                    aria-label="Requested arrival date"
                    aria-invalid={!!validationErrors.requestedArrivalDate}
                  />
                </DeliveryField>
                {requestedArrivalDate && isNotPastDate(requestedArrivalDate) ? (
                  <div style={styles.dateReadout}>
                    <Icon name="clock" size={13} color={C.gold} />
                    <span>Arriving {formatDateDisplay(requestedArrivalDate)}</span>
                  </div>
                ) : null}

                <DeliveryField
                  styles={styles}
                  label="Contact number"
                  required
                  error={validationErrors.contactNumber}
                  icon={<Icon name="phone" size={14} />}
                >
                  <div style={styles.phoneRow} className="p6-grad-row">
                    <span style={styles.prefix} aria-hidden="true">
                      +91
                    </span>
                    <input
                      type="tel"
                      inputMode="numeric"
                      autoComplete="tel-national"
                      style={styles.phoneInput}
                      value={contactNumber}
                      maxLength={10}
                      onChange={handleContactChange}
                      onBlur={handleContactBlur}
                      placeholder="10-digit mobile number"
                      aria-label="Contact number"
                      aria-invalid={!!validationErrors.contactNumber}
                    />
                  </div>
                </DeliveryField>

                <DeliveryField
                  styles={styles}
                  label="Delivery address"
                  required
                  error={validationErrors.deliveryAddress}
                  helper="Use the exact location where the vehicles should deliver the material."
                  icon={<Icon name="location" size={15} />}
                >
                  <textarea
                    className="p6-input p6-grad-input"
                    style={styles.textarea}
                    value={deliveryAddress}
                    onChange={handleAddressChange}
                    onBlur={handleAddressBlur}
                    placeholder="Flat, street, area, landmark, city, state, PIN code"
                    aria-label="Delivery address"
                    aria-invalid={!!validationErrors.deliveryAddress}
                  />
                </DeliveryField>

                <DeliveryField
                  styles={styles}
                  label="Order notes"
                  hint={`${orderNotes.length}/2000`}
                  icon={<Icon name="note" size={14} />}
                >
                  <textarea
                    className="p6-input p6-grad-input"
                    style={styles.textarea}
                    value={orderNotes}
                    maxLength={2000}
                    onChange={handleNotesChange}
                    placeholder="Add access instructions, preferred calling time or site directions."
                    aria-label="Order notes"
                  />
                </DeliveryField>

                <div style={styles.consentWrap}>
                  <label style={styles.consent} className="p6-check-row p6-focusable">
                    <input
                      type="checkbox"
                      checked={confirmed}
                      onChange={(e) => setConfirmed(e.target.checked)}
                      style={styles.consentInput}
                    />
                    <span
                      className="p6-check-box"
                      style={{
                        ...styles.consentBox,
                        ...(confirmed ? styles.consentBoxChecked : {}),
                      }}
                      aria-hidden="true"
                    >
                      {confirmed ? (
                        <Icon name="check" size={13} color="#171006" strokeWidth={3.2} />
                      ) : null}
                    </span>
                    <span style={styles.consentText}>
                      I confirm that the selected material references, vehicle quantities,
                      requested delivery date, contact number and delivery address are correct.
                    </span>
                  </label>
                </div>
              </form>

              <div style={styles.noticeCard}>
                <div style={styles.noticeHead}>
                  <div style={styles.noticeIcon} aria-hidden="true">
                    <Icon name="lock" size={15} />
                  </div>
                  <span style={styles.noticeTitle}>Rate enquiry only</span>
                </div>
                <p style={styles.noticeText}>
                  No payment is required for this rate enquiry. Submitting this request does not
                  confirm delivery — StoneRate will first check material-wise rates and transport
                  availability.
                </p>
              </div>

              <div style={styles.stepsCard}>
                <div style={styles.stepsTitle}>What happens next</div>
                {[
                  "Request submitted to StoneRate.",
                  "StoneRate checks material-wise rates and transport availability.",
                  "Buyer reviews and confirms the rates.",
                ].map((step, i) => (
                  <div
                    style={{
                      ...styles.stepRow,
                      ...(i === 2 ? { paddingBottom: 0 } : {}),
                    }}
                    key={i}
                  >
                    <span style={styles.stepNum}>{i + 1}</span>
                    <span style={styles.stepText}>{step}</span>
                  </div>
                ))}
              </div>
            </>
          ) : null}
        </main>

        {hasMaterials ? (
          <footer style={styles.footer}>
            <div style={styles.progressRow}>
              <div style={styles.progressTrack}>
                <div style={{ ...styles.progressFill, width: `${progressPct}%` }} />
              </div>
              <span style={styles.progressLabel}>{completeness}/5 ready</span>
            </div>
            <div aria-live="polite">
              {submitError ? (
                <div role="alert" style={styles.submitError}>
                  <Icon name="alert" size={15} color={C.red} />
                  <span style={styles.submitErrorText}>{submitError}</span>
                  <button
                    type="button"
                    onClick={() => setSubmitError(null)}
                    style={styles.retryButton}
                    className="p6-focusable"
                  >
                    Dismiss
                  </button>
                </div>
              ) : null}
            </div>
            <button
              type="submit"
              form="rate-request-form"
              disabled={!canSubmit}
              style={submitButtonStyle}
              className="p6-focusable p6-cta"
              aria-busy={isSubmitting}
            >
              {isSubmitting ? (
                <>
                  <span className="p6-spinner" aria-hidden="true" />
                  <span>Submitting Rate Request...</span>
                  <span className="p6-sr-only" role="status">
                    Submitting
                  </span>
                </>
              ) : (
                <>
                  <Icon name="shield" size={16} color="#171006" strokeWidth={2.2} />
                  <span>Submit Rate Request</span>
                </>
              )}
            </button>
            {hasExpiredSample ? (
              <div role="status" style={styles.footerWarn}>
                <Icon name="alert" size={12} color={C.red} />
                <span>Submission disabled — expired sample present.</span>
              </div>
            ) : null}
            <div style={styles.footerNote}>
              <Icon name="lock" size={11} color={C.textDim} />
              <span>Your details are used only for this rate enquiry.</span>
            </div>
          </footer>
        ) : null}
      </div>

      {previewItem ? (
        <ImagePreview
          styles={styles}
          material={previewItem}
          onClose={() => setPreviewItem(null)}
        />
      ) : null}

      {submitted ? (
        <SuccessPopup styles={styles} request={createdRequest} onDone={goToPage7} />
      ) : null}
    </div>
  );
}
